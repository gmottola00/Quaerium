    def tool(
    *,
    namespace: str,
    risk: RiskLevel,
    name: str | None = None,
    description: str | None = None,
    tags: tuple[str, ...] = (),
) -> Callable[[Callable[..., Awaitable[str]]], ToolSpec]:
    """Turn an async function into a ``ToolSpec``.

    Args:
        namespace: Capability name — becomes the prefix of the tool name.
        risk: How the policy gate should treat it. Be honest here; it is the
            only thing standing between the model and an irreversible action.
        name: Override the verb. Defaults to the function name.
        description: Override the description. Defaults to the docstring summary.
        tags: Free-form labels, used by the dashboard to group tools.
    """

    def decorator(fn: Callable[..., Awaitable[str]]) -> ToolSpec:
        if not inspect.iscoroutinefunction(fn):
            raise TypeError(f"{fn.__name__} must be an async function to be a tool")
        summary, _ = split_docstring(fn.__doc__)
        return ToolSpec(
            name=f"{namespace}__{name or fn.__name__}",
            description=description or summary or fn.__name__,
            input_schema=build_schema(fn),
            risk=risk,
            fn=fn,
            tags=tags,
        )

    return decorator

    async def run(self, call_id: str, args: dict[str, Any]) -> ToolResult:
    """Execute and always return a ``ToolResult`` — never raise.

    A tool that blows up must come back to the model as an error result it can
    reason about and recover from, not as an exception that kills the run.
    """
    started = time.monotonic()
    try:
        content = await self.fn(**args)
        ok = True
    except TypeError as exc:
        content = f"Error: invalid arguments for {self.name}: {exc}"
        ok = False
        logger.warning("tool.bad_args", tool=self.name, error=str(exc))
    except Exception as exc:  # noqa: BLE001 - deliberate: the model handles it
        content = f"Error: {type(exc).__name__}: {exc}"
        ok = False
        logger.warning("tool.failed", tool=self.name, error=str(exc), exc_info=True)

    return ToolResult(
        id=call_id,
        name=self.name,
        ok=ok,
        content=content if content else "(no output)",
        duration_ms=(time.monotonic() - started) * 1000,
    )

    async def invoke(self, call: ToolCall) -> ToolResult:
    """Execute a call, returning an error result rather than raising."""
    try:
        spec = self.get(call.name)
    except ToolNotFoundError as exc:
        return ToolResult(id=call.id, name=call.name, ok=False, content=f"Error: {exc}")
    return await spec.run(call.id, call.args)

    def get(self) -> T:
    """Return the client, building it on first use.

    Raises:
        ConfigurationError: Live mode was needed but settings are missing.
            The message names the exact settings to set.
    """
    if self._cached is not None:
        return self._cached

    settings = self._settings()
    if self._fake is not None and self._use_fake(settings):
        logger.info("capability.fake", capability=self._name)
        self._cached = self._fake()
        return self._cached

    missing = self._requires(settings)
    if missing:
        message = f"{self._name} is not configured: set {', '.join(missing)}"
        raise ConfigurationError(f"{message} ({self._hint})" if self._hint else message)

    logger.info("capability.live", capability=self._name)
    self._cached = self._real(settings)
    return self._cached

    def reset(self) -> None:
    """Drop the cached client, closing it best-effort if it has a sync ``close``.

    Used by tests and by settings reloads. A client that needs an async
    close should be closed with :meth:`aclose` instead.
    """
    closer = getattr(self._cached, "close", None)
    if closer is not None and not inspect.iscoroutinefunction(closer):
        try:
            result = closer()
            if inspect.iscoroutine(result):  # pragma: no cover - close() returned a coroutine
                result.close()
        except Exception:  # noqa: BLE001 - teardown must not raise
            logger.warning("capability.close_failed", capability=self._name)
    self._cached = None

    async def aclose(self) -> None:
    """Close the cached client, if it has an async ``aclose``, and drop it."""
    client: Any = self._cached
    if client is not None and hasattr(client, "aclose"):

    def rag_tool(
    pipeline: Any,
    *,
    namespace: str = "kb",
    name: str = "search",
    description: str | None = None,
    top_k: int = 5,
    max_source_chars: int = 500,
) -> ToolSpec:
    """Expose a RAG pipeline as a read-only tool.

    The tool returns the pipeline's answer followed by its numbered sources, so
    the model can quote them. The pipeline's ``run`` is synchronous; it is run
    in a worker thread so the event loop keeps streaming.

    Args:
        pipeline: Anything with ``run(question, *, top_k) -> RagResponse``.
        namespace: Tool namespace; the tool is ``{namespace}__{name}``.
        name: Tool verb.
        description: What the model is told the knowledge base contains. Be
            specific: it is how the model decides to call the tool.
        top_k: Chunks to retrieve.
        max_source_chars: Characters kept per source excerpt.
    """

    async def search(question: str) -> str:
        """Search the knowledge base and answer from it, with numbered sources.

        Args:
            question: A self-contained question; include names and dates from the conversation.
        """
        response = await asyncio.to_thread(pipeline.run, question, top_k=top_k)
        sources = _format_citations(response, max_chars=max_source_chars)
        answer = str(getattr(response, "answer", response))
        return f"{answer}\n\nSources:\n{sources}" if sources else answer

    return tool(namespace=namespace, risk=RiskLevel.READ, name=name, description=description, tags=("rag",))(search)

    def graph_query_tool(
    store: Any,
    *,
    namespace: str = "kb",
    name: str = "graph_query",
    description: str | None = None,
    read_only: bool = True,
    max_rows: int = 50,
) -> ToolSpec:
    """Expose a graph store's native query language (Cypher for Neo4j) as a tool.

    Args:
        store: Anything with ``async query(query, parameters) -> list[dict]``,
            e.g. a :class:`~quaerium.core.graphstore.GraphStoreClient`.
        namespace: Tool namespace.
        name: Tool verb.
        description: What the graph contains: node labels, relationships and
            their properties. The model writes better queries the more it knows.
        read_only: Reject writing clauses before they reach the store, and mark
            the tool ``READ``. With False the tool is ``DESTRUCTIVE``: every
            query waits for a human.
        max_rows: Rows returned to the model; the rest are counted, not shown.
    """

    async def graph_query(query: str, parameters: dict[str, Any] | None = None) -> str:
        """Run a Cypher query against the knowledge graph and return the rows as JSON.

        Args:
            query: The Cypher query. Use $name placeholders for values.
            parameters: Values for the placeholders.
        """
        if read_only and _WRITE_CLAUSE.search(query):
            return "Error: this tool is read-only; the query contains a writing clause."
        rows = await store.query(query, parameters or {})
        shown = rows[:max_rows]
        body = json.dumps(shown, default=str, ensure_ascii=False)
        extra = len(rows) - len(shown)
        return f"{body}\n({extra} more rows not shown)" if extra > 0 else body

    risk = RiskLevel.READ if read_only else RiskLevel.DESTRUCTIVE
    return tool(namespace=namespace, risk=risk, name=name, description=description, tags=("rag", "graph"))(
        graph_query
    )
